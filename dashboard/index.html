      One table feeds every dashboard

      -- sql/06_governed_layer.sql
CREATE TABLE gov_monthly AS
SELECT m.month,
  (SELECT value FROM metric_long
    WHERE metric = 'revenue' AND governed = 1
      AND month = m.month) AS revenue_net_clean,
  (SELECT value FROM metric_long
    WHERE metric = 'customers' AND governed = 1
      AND month = m.month) AS active_customers,
  (SELECT value FROM metric_long
    WHERE metric = 'aov' AND governed = 1
      AND month = m.month) AS aov,
  c.guest_invoices AS guest_buyers
FROM months m JOIN m_customers c USING (month);
